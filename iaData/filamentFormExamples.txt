Output 3:
    $schema = [
        Form\Repeater::make('items')
            ->schema([
                Form\TextInput::make('name')
                    ->label('Item Name')
                    ->required()
                    ->maxLength(255),

                Form\Textarea::make('description')
                    ->label('Item Description')
                    ->rows(2)
                    ->maxLength(500),

                Form\FileUpload::make('src')
                    ->label('Item Image')
                    ->image()
                    ->imageResizeTargetWidth('800')
                    ->imageResizeTargetHeight('600')
                    ->directory('item-images')
                    ->required(),
            ])
            ->label('Items')
            ->addActionLabel('Add Item')
            ->itemLabel(fn (array $state): ?string => $state['name'] ?? null)
            ->collapsible()
            ->cloneable()
            ->columns(2)
    ];
